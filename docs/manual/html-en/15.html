14 Embedding in a C program

Builds on: The language of chapters 1 to 13. Knowledge of C is assumed.

Questions this chapter answers

  • How does a C program open Pinax and call a script?
  • Who decides what a script can do outside itself?
  • How are scripts that are not trusted run?

14.1 Open, register, load, call

A host has four jobs:

  1. give a block of memory and open a state with pinax_open; everything of the state is inside that block;
  2. register the host words scripts may use with pinax_register, with signatures written as in Pinax;
  3. load the script with pinax_load; when it is refused, the reason is in pinax_error_text;
  4. find a word, with its signature, using pinax_find, and call it with pinax_run.

14-host.c

/* A host that lets users write pricing rules in Pinax. The host gives one word (the stock
   level), loads the user's rules, and asks them for a price; a broken rule never takes the
   host down. */
#include "pinax.h"
#include <stdio.h>
#include <string.h>

static int64_t stock[3] = {120, 4, 0};

static pinax_err stock_of(pinax_call *call, void *user, const pinax_args *args) {
    (void)call; (void)user;
    if (args->i[0] < 0 || args->i[0] >= 3)
        return PINAX_ERR_NOT_FOUND;                  /* (item int gives n int or fail) */
    args->ri[0] = stock[args->i[0]];
    return PINAX_OK;
}

static void report(pinax_state *st, const char *what, pinax_err err) {
    pinax_str why = pinax_error_text(st);
    printf("%s: error %#x, %.*s\n", what, (unsigned)err, (int)why.len, (const char *)why.ptr);
}

int main(void) {
    static unsigned char block[256 * 1024];
    pinax_config config = {.struct_size = sizeof config, .region_bytes = 4096, .budget = 10000};
    pinax_state *st;
    pinax_err err = pinax_open((pinax_mem){block, sizeof block}, &config, &st);
    if (err != PINAX_OK) {
        printf("block too small: %zu bytes needed\n", pinax_needed(st));
        return 1;
    }
    static const pinax_word_def words[] = {{"stock_of", "(item int gives n int or fail)", stock_of, NULL, 0}};
    pinax_register(st, "shop", words, 1);

    /* the user's rules: scarce items cost more */
    static const char rules[] =
        "use shop\n"
        "export word price (item int, base int gives p int)\n"
        "  item shop.stock_of or_else (e) 0 end to left\n"
        "  left 0 eq if 0 return end\n"
        "  left 10 lt if base 3 mul 2 div else base end\n"
        "end\n";
    err = pinax_load(st, "rules", (pinax_str){(const unsigned char *)rules, strlen(rules)}, PINAX_LOAD_CODE);
    if (err != PINAX_OK) {
        report(st, "rules refused", err);
        return 1;
    }
    pinax_word price;
    if (pinax_find(st, "price", "(item int, base int gives p int)", 0, &price) != PINAX_OK) {
        report(st, "no price word", PINAX_ERR_NOT_FOUND);
        return 1;
    }
    for (int64_t item = 0; item < 4; ++item) {
        int64_t in[2] = {item, 1000}, out = 0;
        pinax_args args = {.i = in, .ri = &out};
        err = pinax_run(st, price, &args);
        while (err == PINAX_SUSPENDED)
            err = pinax_resume(st, &args);
        if (err != PINAX_OK)
            report(st, "price failed", err);
        else
            printf("item %lld: %lld\n", (long long)item, (long long)out);
    }

    /* a second set of rules with a mistake is refused; the first stays loaded */
    static const char broken[] = "word discount (p int gives int) p \"10\" sub end\n";
    err = pinax_load(st, "broken", (pinax_str){(const unsigned char *)broken, strlen(broken)}, PINAX_LOAD_CODE);
    if (err != PINAX_OK)
        report(st, "broken rules refused", err);
    pinax_close(st);
    return 0;
}

$ ./14-host
item 0: 1000
item 1: 1500
item 2: 0
item 3: 0
broken rules refused: error 0x203, this word takes two int values, or two of the same declared type

Arguments and results pass in arrays by type: integers in args->i[] and args->ri[], strings in args->s[] and args->rs[]. When a host word returns something other than PINAX_OK, a word whose signature has or fail fails (the script takes it with or_else), and a word without it becomes a fault.

Q&A

Q. Can a script delete files or reach the network?

A. Not unless the host registered words that do. The standard modules do not reach outside; everything outside is a host word. Calling C functions with extern reaches only the functions the host’s resolver allows.

14.2 Scripts that are not trusted

config.budget is the number of instructions run at a time. When it is used up the run stops and returns PINAX_SUSPENDED, and the host continues it with pinax_resume or gives up. While a run is stopped, the state takes no other run. To give up, call pinax_abort: it ends the run as a fault would, closing the handles the run opened. What the script stored in globals until then stays.

14-guard.c

/* Running code you do not trust. A budget stops a script that never ends. While a run is
   suspended the state takes no other run; pinax_abort gives the run up, and the state goes on
   with what the script had stored. */
#include "pinax.h"
#include <stdio.h>
#include <string.h>

static unsigned char block[256 * 1024];

int main(void) {
    pinax_config config = {.struct_size = sizeof config, .budget = 50000};
    pinax_state *st;
    if (pinax_open((pinax_mem){block, sizeof block}, &config, &st) != PINAX_OK)
        return 1;
    static const char source[] =
        "global calls int\n"
        "word spin (gives int) calls 1 add to global.calls  0 to n loop n 1 add to n end end\n"
        "word quick (gives int) calls 1 add to global.calls  6 7 mul end\n"
        "word count_calls (gives int) calls end\n";
    if (pinax_load(st, "main", (pinax_str){(const unsigned char *)source, strlen(source)}, PINAX_LOAD_CODE) != PINAX_OK)
        return 1;

    pinax_word w;
    int64_t r = 0;
    pinax_args args = {.ri = &r};
    pinax_find(st, "spin", "(gives int)", 0, &w);
    pinax_err err = pinax_run(st, w, &args);
    int slices = 0;
    while (err == PINAX_SUSPENDED && slices < 5) {     /* the host decides how long to wait */
        ++slices;
        err = pinax_resume(st, &args);
    }
    printf("spin after %d slices: %s\n", slices, err == PINAX_SUSPENDED ? "still running" : "ended");
    pinax_find(st, "quick", "(gives int)", 0, &w);
    printf("another run while it is suspended: %s\n", pinax_run(st, w, &args) == PINAX_ERR_STATE ? "PINAX_ERR_STATE" : "?");

    pinax_abort(st);                                    /* give the spinning run up */
    err = pinax_run(st, w, &args);
    printf("after pinax_abort, quick: %lld\n", (long long)r);
    pinax_find(st, "count_calls", "(gives int)", 0, &w);
    pinax_run(st, w, &args);
    printf("calls stored so far: %lld\n", (long long)r);
    pinax_close(st);
    return 0;
}

Output

$ ./14-guard
spin after 5 slices: still running
another run while it is suspended: PINAX_ERR_STATE
after pinax_abort, quick: 42
calls stored so far: 2

Common mistake: A new pinax_run replaces a stopped run

It does not. A new run is refused with PINAX_ERR_STATE. Either continue the stopped run to its end or give it up with pinax_abort. When even the globals must go back to how they were before the run, save an image before running and open the state again from it (standard 19.9).

14.3 More

Handles (giving a script a C object to hold), binding (host memory as globals), host words that finish later, images and clones, and extern are in clauses 18 and 19 of the standard, with examples.

Summary

  • Give a block and open, register words, load, find and call (standard 19.2–19.5).
  • A failing host word is a failure with or fail, a fault without (19.3).
  • What a script can do outside itself is only what the host registered (13.3, 18).
  • While a run stopped by the budget remains, new runs are refused; go back to an image (19.5, 19.9).
